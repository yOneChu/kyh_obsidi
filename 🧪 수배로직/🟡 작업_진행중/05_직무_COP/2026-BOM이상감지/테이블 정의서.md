| 호기번호 관리할 테이블

| MS-SQL
```SQL
CREATE TABLE PLM_LLM_METADATA (  
    --CATEGORY      VARCHAR(50) PRIMARY KEY,   -- 예: 'PID_QUERY', 'SALES_SPEC', 'DUTY_REVIEW'  
    BRAND        NVARCHAR(100),
    HOGI         NVARCHAR(100),           -- 호기번호
    HOGIOID      NVARCHAR(100),           -- OID
    VERSION      NVARCHAR(50),            -- 마크다운/프롬프트 지침 전문  
    --VERSION       INT DEFAULT 1,        -- 문서 버전  
    --UPDATED_BY    NVARCHAR(50),         -- 최종 수정자 이름  
    CREDATE      DATETIME DEFAULT GETDATE(),   -- 생성일자
    BOMCAL_DATE  NVARCHAR(50)  -- BOM계산날짜
);
```
